Finding HTML Element By ID
1. Javascript
var x = document.getElementById(“menu”);

2. JQuery
var x = $(“#menu”);

Finding HTML Elements By Tag Name
1. Javascript
var x = document.getElementsByTagName(“p”);

2. JQuery
var x = %(“p”);

Finding HTML Elements By Class Name
1. Javascript
var x = document.getElementsByClassName(“post”);

2. JQuery
var x = $(“.post”);

Finding HTML Elements By CSS Selectors
1. Javascript
var x = document.querySelectorAll(“p.post”);

2. JQuery
var x = $(“p.post”);

JQuery Events

1. $(document).ready() - executes a function when the document is fully loaded

$(document).ready(function() {
  // code here
});

2. click() - executes when the user clicks on the HTML element

$(document).ready(function(){
	$(“p”).click(function() {
		$(this).css(“color”, “red”);
	});
});

3. dblclick() - executes when the user double-clicks on the HTML element

$(document).ready(function(){
	$(“p”).dblclick(function() {
		$(this).css(“color”, “red”);
	});
});

4. mouseenter() - executes when the mouse pointer enters the HTML element

$(document).ready(function(){
	$(“p”).mouseenter(function() {
		$(this).text("Hello World!");
	});
});

5. mouseleave() - executes when the mouse pointer leaves the HTML element

$(document).ready(function(){
	$(“p”).mouseleave(function() {
		$(this).text(“Bye World!”);
	});
});


6. mousedown() - executes when the mouse button is pressed down on the HTML element

$(document).ready(function(){
	$(“p”).mousedown(function() {
		$(this).text(“Mouse down!”);
	});
});

7. mouseup() - executes when the mouse button is released over the HTML element

$(document).ready(function(){
	$(“p”).mouseup(function() {
		$(this).text(“Mouse up!”);
	});
});

8. hover() - combined mouseenter() and mouseleave() methods

$(document).ready(function(){
	$(“p”).hover(function() {
		$(this).text(“You entered!”);
	}, function(){
		$(this).text(“You exited!”);
	});
})

9. focus() - executes when the form field gets focus

$(document).ready(function(){
	$(“input”).focus(function(){
		$(this).css(“background-color”, “gray”);
	});
});

10. blur() - executes when the form field loses focus

$(document).ready(function(){
	$(“input”).blur(function(){
		$(this).css(“background-color”, “white”);
	});
});

11. on() - attaches one or more event handlers for the selected elements

$(“p”).on({
	mouseenter: function() {
		$(this).css(“color”, “red”);
	}, mouseleave: function() {
		$(this).css(“color”, “black”);
	}
});

Others
1. append() - inserts content at the end of the selected elements
2. prepend() - inserts content at the beginning of the selected elements
3. after() - inserts content after the selected elements
4. before() - inserts content before the selected elements
5. remove() - removes the selected elements
6. empty() - removes the child elements of the selected elements
7. clone() - creates a copy of the selected elements
